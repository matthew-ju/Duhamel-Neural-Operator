% a slab of the device: bottom z0, height h, front/side/top fill colour
\newcommand{\slab}[4][black!60]{%
  \fill[#4] (0,#2) rectangle (\Lx,#2+#3);
  \fill[#4!85!black] (\Lx,#2) -- ++(\Dx,\Dy) -- ++(0,#3) -- (\Lx,#2+#3) -- cycle;
  \fill[#4!92!white] (0,#2+#3) -- (\Lx,#2+#3) -- ++(\Dx,\Dy) -- ++(-\Lx,0) -- cycle;
  \draw[#1,line width=0.35pt,line join=round] (0,#2) rectangle (\Lx,#2+#3)
     (\Lx,#2) -- ++(\Dx,\Dy) -- ++(0,#3) -- ++(-\Lx,0) -- (0,#2+#3) (\Lx,#2+#3) -- ++(\Dx,\Dy);
}
